// ================= QUERY: Kobo_Raw =================
let
    // Kobo export setting "EPI_BACKUP" (same layout as the downloaded EPI_Kobo_Data.xlsx)
    Source = Excel.Workbook(Web.Contents("https://eu.kobotoolbox.org/api/v2/assets/a5aJnksV6mK3Fgese2DPMy/export-settings/esKvGvapbsM4y4zohJ2AuRL/data.xlsx"), true, true),
    Data = Table.SelectRows(Source, each [Kind] = "Sheet"){0}[Data],
    Dates = Table.TransformColumnTypes(Data, {{"Date of Visit", type date}, {"Reporting Year and Month", type date}, {"When was the last session?", type date}}, "en-US"),
    Clean = Table.ReplaceErrorValues(Dates, {{"Date of Visit", null}, {"Reporting Year and Month", null}, {"When was the last session?", null}})
in
    Clean

// ================= QUERY: fnYesNo =================
(tbl as table, pairs as list) as table =>
    List.Accumulate(pairs, tbl, (t, p) =>
        let
            ans = (r) => Text.Lower(Text.Trim(Text.From(Record.Field(r, p[src]) ?? ""))),
            t1 = Table.AddColumn(t, p[yes], each if ans(_) = "yes" then 1 else 0, Int64.Type),
            t2 = Table.AddColumn(t1, p[no], each if ans(_) = "no" then 1 else 0, Int64.Type),
            t3 = if p[total] = null then t2 else Table.AddColumn(t2, p[total], (r) => Record.Field(r, p[yes]) + Record.Field(r, p[no]), Int64.Type)
        in
            t3)
